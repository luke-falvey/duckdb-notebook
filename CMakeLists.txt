cmake_minimum_required(VERSION 3.5)

set(TARGET_NAME notebook)

set(EXTENSION_NAME ${TARGET_NAME}_extension)
set(LOADABLE_EXTENSION_NAME ${TARGET_NAME}_loadable_extension)

project(${TARGET_NAME})

set(CMAKE_CXX_STANDARD
    "17"
    CACHE STRING "C++ standard to enforce")
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include_directories(src/include)
# Vendored with DuckDB: cpp-httplib for the server, miniz for gzip in exports.
include_directories(${CMAKE_SOURCE_DIR}/third_party/httplib ${CMAKE_SOURCE_DIR}/third_party/miniz)

# The browser frontend (frontend/*) is compiled into the extension as byte arrays, so nothing is
# fetched at run time. CMake re-runs when any of the files changes.
file(GLOB NOTEBOOK_ASSETS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/frontend/*)
set(NOTEBOOK_ASSET_CPP ${CMAKE_CURRENT_BINARY_DIR}/notebook_assets.cpp)
set(NOTEBOOK_ASSET_CODE "// Generated by CMakeLists.txt from frontend/. Do not edit.\n#include \"notebook/session.hpp\"\n#include <unordered_map>\n\nnamespace duckdb {\n\nnamespace {\n\n")
set(NOTEBOOK_ASSET_TABLE "")
foreach(asset ${NOTEBOOK_ASSETS})
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${asset})
  get_filename_component(asset_name ${asset} NAME)
  string(MAKE_C_IDENTIFIER ${asset_name} asset_id)
  file(READ ${asset} asset_hex HEX)
  string(LENGTH "${asset_hex}" asset_hex_length)
  math(EXPR asset_size "${asset_hex_length} / 2")
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," asset_hex "${asset_hex}")
  string(APPEND NOTEBOOK_ASSET_CODE "const unsigned char ASSET_${asset_id}[] = {${asset_hex}0x00};\n")
  string(APPEND NOTEBOOK_ASSET_TABLE "\t\t    {\"${asset_name}\", string(reinterpret_cast<const char *>(ASSET_${asset_id}), ${asset_size})},\n")
endforeach()
string(APPEND NOTEBOOK_ASSET_CODE "\n} // namespace\n\nconst string &GetAsset(const string &name) {\n\tstatic const std::unordered_map<string, string> assets = {\n${NOTEBOOK_ASSET_TABLE}\t};\n\tstatic const string missing;\n\tauto found = assets.find(name);\n\treturn found == assets.end() ? missing : found->second;\n}\n\n} // namespace duckdb\n")
file(WRITE ${NOTEBOOK_ASSET_CPP}.tmp "${NOTEBOOK_ASSET_CODE}")
configure_file(${NOTEBOOK_ASSET_CPP}.tmp ${NOTEBOOK_ASSET_CPP} COPYONLY)

# core/      plain C++ with no DuckDB dependency: format, lexer, script generation, graph.
#            Unit-tested on its own with `make core_test`.
# functions/ the SQL surface, one file per function, written against DuckDB's C++ API.
set(EXTENSION_SOURCES
    src/core/sql_lexer.cpp
    src/core/format.cpp
    src/core/script.cpp
    src/core/graph.cpp
    src/core/json.cpp
    src/core/yaml.cpp
    src/core/chart.cpp
    src/core/cache.cpp
    src/core/document.cpp
    src/functions/common.cpp
    src/functions/references.cpp
    src/functions/session.cpp
    src/functions/result_cache.cpp
    src/functions/notebook_cells.cpp
    src/functions/notebook_sql.cpp
    src/functions/notebook_run.cpp
    src/functions/notebook_chart_query.cpp
    src/functions/storage.cpp
    src/functions/export.cpp
    src/functions/server.cpp
    src/functions/peek.cpp
    ${NOTEBOOK_ASSET_CPP}
    src/notebook_extension.cpp)

build_static_extension(${TARGET_NAME} ${EXTENSION_SOURCES})
build_loadable_extension(${TARGET_NAME} " " ${EXTENSION_SOURCES})

install(
  TARGETS ${EXTENSION_NAME}
  EXPORT "${DUCKDB_EXPORT_SET}"
  LIBRARY DESTINATION "${INSTALL_LIB_DIR}"
  ARCHIVE DESTINATION "${INSTALL_LIB_DIR}")
